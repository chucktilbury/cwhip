982 (void)self;

984 if (hostname == NULL || port < 0 || port > 65535 || backlog < 0 ||
985 backlog > INT_MAX) {

987 free(hostname);

989 }

991 (void)snprintf(service, sizeof(service), "%d", (int)port);
992 struct addrinfo hints;

994 hints.ai_family = AF_INET;
995 hints.ai_socktype = SOCK_STREAM;
996 hints.ai_flags = AI_PASSIVE;
997 struct addrinfo *addresses = NULL;
998 const int lookup = getaddrinfo(hostname[0] == '\0' ? NULL : hostname,

1003 return -1;

1008 address->ai_protocol);
1009 if (candidate < 0) continue;

1013 listen(candidate, backlog > 0 ? (int)backlog : 16) == 0) {
1014 server_fd = candidate;
1015 break;

1017 (void)close(candidate);
1018 }
1019 const int saved_error = errno;
1020 freeaddrinfo(addresses);
1021 if (server_fd < 0) cwhip_runtime_set_error(saved_error == 0 ? EIO : saved_error);
1022 else cwhip_runtime_clear_error();
