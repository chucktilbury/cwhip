131void dumpStatement(const Statement& statement, std::ostream& output, int depth) {
132 indent(output, depth);
133 switch (statement.kind) {
134 case StatementKind::Declaration:
135 output << "Declaration [" << statement.declaredType << " " << statement.name << "]\n";
136 break;
137 case StatementKind::Assignment:
138 output << "Assignment";
139 if (statement.assignmentOperator != "=") {
140 output << " [" << statement.assignmentOperator << "]";
141 }
142 output << '\n';
143 break;
144 case StatementKind::Print: output << "Print\n"; break;
145 case StatementKind::If: output << "If\n"; break;
146 case StatementKind::While: output << "While\n"; break;
147 case StatementKind::DoWhile: output << "Do while\n"; break;
148 case StatementKind::Break: output << "Break\n"; break;
149 case StatementKind::Continue: output << "Continue\n"; break;
150 case StatementKind::ForEach:
151 output << (statement.keyName.empty()
152 ? "For each [" + statement.name + "]\n"
153 : "For each [" + statement.keyName + ", " + statement.name + "]\n");
154 break;
155 case StatementKind::Block: output << "Block\n"; break;
156 case StatementKind::Return: output << "Return\n"; break;
157 case StatementKind::Expression: output << "Expression statement\n"; break;
158 case StatementKind::SuperConstructorCall:
159 output << (statement.virtualBaseInitializer ? "Super virtual constructor [" :
160 "Super constructor [")
161 << statement.name << "]\n";
162 break;
163 case StatementKind::Raise: output << "Raise\n"; break;
164 case StatementKind::Try: output << "Try\n"; break;
165 case StatementKind::InlineC: output << "Inline C\n"; break;
166 }
167 if (statement.target) {
168 dumpExpression(*statement.target, output, depth + 1);
169 }
170 for (const auto& expression : statement.expressions) {
171 dumpExpression(*expression, output, depth + 1);
172 }
173 if (!statement.body.empty()) {
174 dumpStatements(statement.body, output, depth + 1);
175 }
176 if (statement.kind == StatementKind::Try) {
177 for (const auto& handler : statement.exceptionHandlers) {
178 indent(output, depth + 1);
179 output << "Except";
180 if (!handler.exceptionType.empty()) {
181 output << " [" << handler.exceptionType << "]";
182 }
183 if (handler.hasBinding) output << " as [" << handler.name << "]";
184 output << '\n';
185 dumpStatements(handler.body, output, depth + 2);
186 }
187 } else if (statement.hasAlternate) {
188 indent(output, depth + 1);
189 output << "Else\n";
190 dumpStatements(statement.alternate, output, depth + 2);
191 }
192 if (statement.hasCleanup) {
193 indent(output, depth + 1);
194 output << "Finally\n";
195 dumpStatements(statement.cleanup, output, depth + 2);
196 }
197}

207void dumpAst(const Program& program, std::ostream& output) {
208 output << "Program\n";
209 for (const auto& import : program.imports) {
210 indent(output, 1);
211 output << "Import [" << import.moduleName;
212 if (!import.alias.empty()) output << " as " << import.alias;
213 output << "]\n";
214 }
215 for (const auto& declaration : program.namespaces) {
216 indent(output, 1);
217 output << "Namespace [";
218 for (std::size_t index = 0; index < declaration.path.size(); ++index) {
219 if (index != 0) output << ".";
220 output << declaration.path[index];
221 }
222 output << "]\n";
223 }
224 for (const auto& definition : program.outOfLineMethods) {
225 indent(output, 1);
226 output << "Out-of-line method [" << definition.className << "."
227 << definition.method.returnType << " " << definition.method.name << "(";
228 for (std::size_t index = 0; index < definition.method.parameters.size(); ++index) {
229 if (index != 0) output << ", ";
230 output << definition.method.parameters[index].type << " "
231 << definition.method.parameters[index].name;
232 }
233 output << ")";
234 if (definition.method.externalBinding) {
235 output << " from \"" << definition.method.externalSymbol << "\"";
236 }
237 output << "]\n";
238 }
239 for (const auto& declaration : program.classes) {
240 if (declaration.builtin) continue;
241 indent(output, 1);
242 output << "Class [" << declaration.name;
243 if (!declaration.baseClassNames.empty()) {
244 output << " : ";
245 for (std::size_t index = 0; index < declaration.baseClassNames.size(); ++index) {
246 if (index != 0) output << ", ";
247 const char* access = declaration.baseAccess[index] == AccessLevel::Public
248 ? "public "
249 : declaration.baseAccess[index] == AccessLevel::Protected
250 ? "protected "
251 : "private ";
252 output << access;
253 if (declaration.baseVirtual[index]) output << "virtual ";
254 output << declaration.baseClassNames[index];
255 }
256 }
257 output << "]\n";
258 for (const auto& field : declaration.fields) {
259 indent(output, 2);
260 output << "Field [" << field.type << " " << field.name << "]\n";
261 }
262 for (const auto& member : declaration.enumMembers) {
263 indent(output, 2);
264 output << "Enum member [" << member.name;
265 if (member.value) output << " = " << *member.value;
266 output << "]\n";
267 }
268 for (const auto& method : declaration.methods) {
269 indent(output, 2);
270 output << (method.constructor ? "Constructor [" :
271 method.declarationOnly ? "Method declaration [" : "Method [")
272 << method.returnType << " " << method.name << "]\n";
273 dumpStatements(method.body, output, 3);
274 }
275 }
276 dumpStatements(program.statements, output, 1);
277}
