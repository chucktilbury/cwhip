16std::string llvmBytes(const std::string& bytes) {
17 std::ostringstream escaped;
18 for (const unsigned char byte : bytes) {
19 if (byte >= 0x20 && byte <= 0x7e && byte != '"' && byte != '\\') {
20 escaped << static_cast<char>(byte);
21 } else {
22 escaped << '\\' << std::uppercase << std::hex << std::setw(2) << std::setfill('0')
23 << static_cast<unsigned>(byte) << std::nouppercase << std::dec;
24 }
25 }
26 return escaped.str();
27}
