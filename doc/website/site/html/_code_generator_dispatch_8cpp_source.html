37const MethodDeclaration* CodeGenerator::findMethod(const ClassDeclaration& owner,
38 const std::string& name) const {
39 std::unordered_set<std::string> seenVirtual;
40 const auto visit = [this, &name, &seenVirtual](
41 const auto& self, const ClassDeclaration& current)
42 -> const MethodDeclaration* {
43 for (const auto& method : current.methods) {
44 if (method.name == name && !method.constructor && !method.destructor) return &method;
45 }
46 const MethodDeclaration* result = nullptr;
47 for (std::size_t index = 0; index < current.baseClassNames.size(); ++index) {
48 const auto& baseName = current.baseClassNames[index];
49 const auto base = classes_.find(baseName);
50 if (base == classes_.end()) continue;
51 if (current.baseVirtual[index] && !seenVirtual.emplace(baseName).second) continue;
52 if (const auto* candidate = self(self, *base->second)) {
53 if (result != nullptr) return nullptr;
54 result = candidate;
55 }
56 }
57 return result;
58 };
59 return visit(visit, owner);
60}

63 const std::string& name,

90std::vector<const MethodDeclaration*> CodeGenerator::methodSlots(
91 const ClassDeclaration& owner) const {
92 std::vector<const MethodDeclaration*> slots;
93 const auto append = [this, &slots](const auto& self,
94 const ClassDeclaration& declaration) -> void {
95 for (const auto& baseName : declaration.baseClassNames) {
96 const auto base = classes_.find(baseName);
97 if (base != classes_.end()) self(self, *base->second);
98 }
99 for (const auto& method : declaration.methods) {
100 if (method.constructor || method.destructor) continue;
101 const auto inherited = std::find_if(
102 slots.begin(), slots.end(), [&method](const MethodDeclaration* candidate) {
103 return candidate->name == method.name &&
104 sameParameterTypes(*candidate, method);
105 });
106 if (inherited == slots.end()) {
107 slots.push_back(&method);
108 } else {
109 *inherited = &method;
110 }
111 }
112 };
113 append(append, owner);
114 return slots;
115}
