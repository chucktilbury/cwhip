70static int make_pipe(int descriptors[2]) {
71 if (pipe(descriptors) != 0) return 0;
72 const int read_flags = fcntl(descriptors[0], F_GETFL);
73 const int close_read = fcntl(descriptors[0], F_GETFD);
74 const int close_write = fcntl(descriptors[1], F_GETFD);
75 if (read_flags < 0 || close_read < 0 || close_write < 0 ||
76 fcntl(descriptors[0], F_SETFL, read_flags | O_NONBLOCK) != 0 ||
77 fcntl(descriptors[0], F_SETFD, close_read | FD_CLOEXEC) != 0 ||
78 fcntl(descriptors[1], F_SETFD, close_write | FD_CLOEXEC) != 0) {
79 const int saved = errno;
80 (void)close(descriptors[0]);
81 (void)close(descriptors[1]);
82 errno = saved;
83 return 0;
84 }

202static int append_output(char **data, size_t *length, size_t *capacity,
203 const char *bytes, size_t count) {
204 if (*length > (size_t)INT64_MAX || count > (size_t)INT64_MAX - *length) {
205 errno = EOVERFLOW;
206 return 0;
207 }
208 const size_t required = *length + count;
209 if (required > *capacity) {
210 size_t grown = *capacity == 0 ? 4096 : *capacity;
211 while (grown < required) {
212 if (grown > (size_t)INT64_MAX / 2) {
213 grown = (size_t)INT64_MAX;
214 break;
215 }
216 grown *= 2;
217 }
218 char *next = (char *)realloc(*data, grown);
219 if (next == NULL) {
220 errno = ENOMEM;
221 return 0;
222 }
223 *data = next;
224 *capacity = grown;
225 }
226 memcpy(*data + *length, bytes, count);
227 *length = required;
228 return 1;

231static void drain_pipe(CwhipProcess *process, int *descriptor, int is_error) {
232 char chunk[8192];
233 for (;;) {
234 const ssize_t count = read(*descriptor, chunk, sizeof(chunk));
235 if (count > 0) {
236 int stored;
237 if (is_error) {
238 if (process->stderr_error != 0) continue;
239 stored = append_output(&process->stderr_bytes, &process->stderr_length,
240 &process->stderr_capacity, chunk, (size_t)count);
241 } else {
242 if (process->stdout_error != 0) continue;
243 stored = append_output(&process->stdout_bytes, &process->stdout_length,
244 &process->stdout_capacity, chunk, (size_t)count);
245 }
246 if (!stored) {
247 if (is_error) process->stderr_error = errno;
248 else process->stdout_error = errno;
249 if (process->output_error == 0) process->output_error = errno;
250 }
251 continue;
252 }
253 if (count == 0) {
254 close_fd(descriptor);
255 return;
256 }
257 if (errno == EINTR) continue;
258 if (errno == EAGAIN || errno == EWOULDBLOCK) return;
259 process->output_error = errno;
260 close_fd(descriptor);
261 return;
262 }

266 (void)self;
267 CwhipProcess *process = (CwhipProcess *)object;
268 if (process == NULL) {
269 cwhip_runtime_set_error(EINVAL);
270 return 0;
271 }
272 if (process->waited) return 1;
273 int child_reaped = 0;
274 int wait_status = 0;
275 int wait_error = 0;
276 cwhip_runtime_gil_release();
277 while (!child_reaped || process->stdout_fd >= 0 || process->stderr_fd >= 0) {
278 struct pollfd descriptors[2] = {
279 {process->stdout_fd, POLLIN | POLLHUP, 0},
280 {process->stderr_fd, POLLIN | POLLHUP, 0}
281 };
282 (void)poll(descriptors, 2, 20);
283 if (process->stdout_fd >= 0) drain_pipe(process, &process->stdout_fd, 0);
284 if (process->stderr_fd >= 0) drain_pipe(process, &process->stderr_fd, 1);
285 if (!child_reaped) {
286 const pid_t waited = waitpid(process->pid, &wait_status, WNOHANG);
287 if (waited == process->pid) child_reaped = 1;
288 else if (waited < 0 && errno != EINTR) {
289 wait_error = errno;
290 child_reaped = 1;
291 close_fd(&process->stdout_fd);
292 close_fd(&process->stderr_fd);
293 }
294 }
295 }
296 if (!child_reaped) {
297 while (waitpid(process->pid, &wait_status, 0) < 0) {
298 if (errno == EINTR) continue;
299 wait_error = errno;
300 break;
301 }
302 }
303 cwhip_runtime_gil_acquire();
304 if (wait_error != 0) {
305 cwhip_runtime_set_error(wait_error);
306 return 0;
307 }
308 process->status = WIFEXITED(wait_status)
309 ? WEXITSTATUS(wait_status)
310 : WIFSIGNALED(wait_status) ? -WTERMSIG(wait_status) : -1;
311 process->waited = 1;
312 if (process->output_error != 0) cwhip_runtime_set_error(process->output_error);
313 else cwhip_runtime_clear_error();
314 return 1;
315}
