159 std::size_t resultValue = 0;

163 throw std::invalid_argument(optionName + " requires a non-negative integer");

310int runCompiler(const BuildContext& context, const std::vector<std::string>& arguments,
311 const std::string& operation) {
312 std::string command = shellQuote(context.compiler.string());
313 if (CWHIP_COVERAGE &&
314 std::find(arguments.begin(), arguments.end(), "-r") == arguments.end()) {
315 command += " -fprofile-instr-generate -fcoverage-mapping";
316 }
317 if (std::string(CWHIP_SANITIZER_LIST).size() != 0 &&
318 std::find(arguments.begin(), arguments.end(), "-r") == arguments.end()) {
319 command += " " + shellQuote("-fsanitize=" CWHIP_SANITIZER_LIST);
320 command += " -fno-omit-frame-pointer -g";
321 }
322 for (const auto& argument : arguments) command += " " + shellQuote(argument);
323 if (context.verbosity >= pathVerbosity) std::cerr << "[command] " << command << '\n';
324 const int status = std::system(command.c_str());
325 if (status == -1) {
326 std::cerr << "cwhip: could not start compiler " << context.compiler.string() << '\n';
327 return 1;
328 }
329 if (!WIFEXITED(status) || WEXITSTATUS(status) != 0) {
330 const int exitCode = WIFEXITED(status) ? WEXITSTATUS(status) : 128;
331 std::cerr << "cwhip: compiler failed to " << operation
332 << " (exit status " << exitCode << ")\n";
333 return 1;
334 }
335 return 0;
336}

351 const std::vector<std::string>& objectPaths,
352 const std::string& inlineShimPath,
353 const std::vector<std::string>& packageLibraryPaths,
354 const std::vector<std::string>& packageLibraries,
355 const std::vector<std::string>& libraryPaths,

379 for (const auto& library : libraries) arguments.push_back("-l" + library);

400void writeLinkSidecar(const std::string& objectPath, const LinkInputs& inputs) {

419LinkInputs readLinkSidecars(const std::vector<std::string>& objectPaths) {
420 LinkInputs inputs;
421 for (const auto& objectPath : objectPaths) {
422 const auto path = linkSidecarPath(objectPath);
423 std::ifstream input(path);
424 if (!input) {
425 if (!std::filesystem::exists(path)) continue;
426 throw std::runtime_error("cannot read package link metadata: " + path);
427 }
428 std::string line;
429 if (!std::getline(input, line) || line != "CWHIP-LINK 1") {
430 throw std::runtime_error("invalid package link metadata header: " + path);
431 }
432 std::size_t lineNumber = 1;
433 while (std::getline(input, line)) {
434 ++lineNumber;
435 if (line.empty()) continue;
436 std::istringstream row(line);
437 char kind = '\0';
438 std::string value;
439 std::string trailing;
440 if (!(row >> kind >> std::quoted(value)) || (row >> trailing) ||
441 (kind != 'L' && kind != 'l') || value.empty()) {
442 throw std::runtime_error(path + ":" + std::to_string(lineNumber) +
443 ": invalid package link metadata row");
444 }
445 if (kind == 'L') {
446 appendUnique(inputs.libraryPaths, value);
447 } else {
448 if (value.front() == '-') {
449 throw std::runtime_error(path + ":" +
450 std::to_string(lineNumber) +
451 ": invalid library name");
452 }
453 appendUnique(inputs.libraries, value);
454 }
455 }
456 }
457 return inputs;
458}

460int buildObject(const BuildContext& context, const std::vector<std::string>& irPaths,
461 const std::string& inlineShimPath,
462 const std::string& outputPath,
463 std::vector<std::string>& temporaryPaths, bool debug) {
464 ensureOutputDirectory(outputPath);
465 std::vector<std::string> objectPaths;
466 for (std::size_t index = 0; index < irPaths.size(); ++index) {
467 const auto objectPath = temporaryPath(outputPath, "ir." + std::to_string(index));
468 temporaryPaths.push_back(objectPath);
469 std::vector<std::string> arguments{
470 "-Wno-override-module", "-x", "ir", irPaths[index], "-c", "-o", objectPath};
471 if (debug) arguments.push_back("-g");
472 if (runCompiler(context, arguments, "compile LLVM IR to an object") != 0) return 1;
473 objectPaths.push_back(objectPath);
474 }
475 if (!inlineShimPath.empty()) {
476 const auto shimObjectPath = temporaryPath(outputPath, "shim");
477 temporaryPaths.push_back(shimObjectPath);
478 std::vector<std::string> arguments{
479 "-I", context.includeDirectory.string(), "-x", "c", inlineShimPath,
480 "-c", "-o", shimObjectPath};
481 if (debug) arguments.push_back("-g");
482 if (runCompiler(context, arguments, "compile an inline C shim") != 0) return 1;
483 objectPaths.push_back(shimObjectPath);
484 }
485 std::vector<std::string> arguments{"-r"};
486 if (debug) arguments.push_back("-g");
487 arguments.insert(arguments.end(), objectPaths.begin(), objectPaths.end());
488 arguments.insert(arguments.end(), {"-o", outputPath});
489 return runCompiler(context, arguments, "combine relocatable objects");
490}

492bool samePath(const std::string& left, const std::string& right) {
493 std::error_code error;

495 if (error) leftPath = std::filesystem::absolute(left).lexically_normal();
496 error.clear();

498 if (error) rightPath = std::filesystem::absolute(right).lexically_normal();
