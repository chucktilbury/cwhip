18std::string subobjectTag(const std::vector<std::string>& path,
19 const ClassDeclaration&) {
20 if (path.empty()) return "root";
21 std::string tag = "p";
22 for (std::size_t index = 0; index < path.size(); ++index) {
23 if (tag.size() > 1) tag += "_";
24 tag += path[index];
25 }
26 return tag;
27}

36std::string viewMetadataName(const ClassDeclaration& owner,
37 const std::vector<std::string>& path) {
38 if (owner.builtin && path.empty()) return "@cwhip_exception_class_meta";
39 // The runtime creates String instances itself (exception messages), so the
40 // String metadata is a single externally visible object shared by every
41 // module instead of a per-module private constant.
42 if (owner.name == "String" && path.empty()) return "@cwhip_string_class_meta";
43 return path.empty() ? "@.cw.class.meta." + owner.name
44 : "@.cw.view.meta." + owner.name + "." +
45 subobjectTag(path, owner);
