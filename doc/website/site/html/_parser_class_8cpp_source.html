26ClassDeclaration Parser::parseClass() {
27 const auto keyword = consume(TokenType::Class, "'class'");
28 const auto name = consume(TokenType::Identifier, "class name");
29 ClassDeclaration declaration;
30 declaration.name = name.text;
31 declaration.namespacePath = namespacePath_;
32 declaration.location = keyword.location;
33 if (match(TokenType::Colon)) {
34 do {
35 AccessLevel access = AccessLevel::Public;
36 bool isVirtual = false;
37 for (;;) {
38 if (match(TokenType::Public)) access = AccessLevel::Public;
39 else if (match(TokenType::Protected)) access = AccessLevel::Protected;
40 else if (match(TokenType::Private)) access = AccessLevel::Private;
41 else if (match(TokenType::Virtual)) {
42 if (isVirtual) error(previous(), "duplicate 'virtual' base modifier");
43 isVirtual = true;
44 } else {
45 break;
46 }
47 }
48 const auto baseLocation = current().location;
49 const auto baseName = parseQualifiedIdentifier("base class name");
50 declaration.baseClassNames.push_back(baseName);
51 declaration.baseLocations.push_back(baseLocation);
52 declaration.baseAccess.push_back(access);
53 declaration.baseVirtual.push_back(isVirtual);
54 if (declaration.baseClassNames.size() == 1) {
55 declaration.baseClassName = baseName;
56 declaration.baseLocation = baseLocation;
57 }
58 } while (match(TokenType::Comma));
59 }
60 consume(TokenType::LeftBrace, "'{' after class name");
61 AccessLevel memberAccess = AccessLevel::Public;
62 while (!check(TokenType::RightBrace) && !check(TokenType::End)) {
63 skipNewlines();
64 if (check(TokenType::RightBrace) || check(TokenType::End)) {
65 break;
66 }
67 if (check(TokenType::Import)) {
68 error(current(), "'import' is only allowed at top level");
69 }
70 if (check(TokenType::Include)) {
71 error(current(), "'include' is only allowed at top level");
72 }
73 if (check(TokenType::Inline)) {
74 error(current(), "'inline' is only allowed inside a function or method body");
75 }
76 if (isEnumKeyword(current()) &&
77 current_ + 1 < tokens_.size() &&
78 tokens_[current_ + 1].type == TokenType::LeftBrace) {
79 ++current_;
80 parseEnumMembers(declaration, memberAccess);
81 continue;
82 }
83 if ((check(TokenType::Public) || check(TokenType::Protected) ||
84 check(TokenType::Private)) &&
85 current_ + 1 < tokens_.size() &&
86 tokens_[current_ + 1].type == TokenType::Colon) {
87 if (match(TokenType::Public)) memberAccess = AccessLevel::Public;
88 else if (match(TokenType::Protected)) memberAccess = AccessLevel::Protected;
89 else {
90 match(TokenType::Private);
91 memberAccess = AccessLevel::Private;
92 }
93 consume(TokenType::Colon, "':' after access section");
94 continue;
95 }
96 if (check(TokenType::Identifier) && current().text == "destroy" &&
97 current_ + 1 < tokens_.size() &&
98 (tokens_[current_ + 1].type == TokenType::LeftBrace ||
99 tokens_[current_ + 1].type == TokenType::LeftParen)) {
100 const auto destructorName = current();
101 if (tokens_[current_ + 1].type != TokenType::LeftBrace) {
102 error(destructorName, "destructors use 'destroy { ... }' syntax");
103 }
104 ++current_;
105 MethodDeclaration destructor;
106 destructor.name = destructorName.text;
107 destructor.returnType = "void";
108 destructor.location = destructorName.location;
109 destructor.destructor = true;
110 destructor.access = memberAccess;
111 destructor.body = parseBlock();
112 declaration.methods.push_back(std::move(destructor));
113 continue;
114 }
115 const auto type = current();
116 const auto typeName = parseType(true);
117 if (type.type == TokenType::Identifier && typeName == declaration.name &&
118 check(TokenType::LeftParen)) {
119 auto constructor = parseMethod(type, type, true);
120 constructor.access = memberAccess;
121 declaration.methods.push_back(std::move(constructor));
122 continue;
123 }
124 const auto member = consume(TokenType::Identifier, "field or method name");
125 if (check(TokenType::LeftParen)) {
126 if (member.text == "destroy") {
127 error(member, "destructors use 'destroy { ... }' syntax");
128 }
129 if (member.text == declaration.name) {
130 error(member, "constructor '" + member.text +
131 "' must omit the return type; write '" + member.text +
132 "(...)' instead of '" + typeName + " " + member.text +
133 "(...)'");
134 }
135 auto method = parseMethod(type, member, false, typeName);
136 method.access = memberAccess;
137 declaration.methods.push_back(std::move(method));
138 } else {
139 if (typeName == "void") error(type, "fields cannot have type void");
140 consumeStatementTerminator();
141 declaration.fields.push_back({typeName, member.text, type.location, memberAccess});
142 }
143 }
144 consume(TokenType::RightBrace, "'}' after class body");
145 return declaration;
146}

148void Parser::parseEnumMembers(ClassDeclaration& declaration, AccessLevel access) {
149 const auto blockIndex = declaration.enumBlockCount++;
150 consume(TokenType::LeftBrace, "'{' after 'enum'");
151 skipNewlines();
152 while (!check(TokenType::RightBrace) && !check(TokenType::End)) {
153 const auto name = consume(TokenType::Identifier, "enum member name");
154 EnumMemberDeclaration member;
155 member.name = name.text;
156 member.location = name.location;
157 member.access = access;
158 member.blockIndex = blockIndex;
159 if (match(TokenType::Equal)) {
160 member.initializer = parseExpression();
161 }
162 declaration.enumMembers.push_back(std::move(member));
163
164 if (match(TokenType::Comma)) {
165 skipNewlines();
166 if (check(TokenType::RightBrace)) break;
167 continue;
168 }
169 if (check(TokenType::RightBrace)) break;
170 if (!check(TokenType::Newline)) {
171 error(current(), "expected ',' or newline after enum member");
172 }
173 skipNewlines();
174 }
175 consume(TokenType::RightBrace, "'}' after enum members");
176}

178OutOfLineMethodDefinition Parser::parseOutOfLineMethodDefinition() {
179 const auto returnType = current();
180 const auto returnTypeName = parseType(true);
181 std::string className = consume(TokenType::Identifier, "class name before '.'").text;
182 consume(TokenType::Dot, "'.' between class and method name");
183 auto member = consume(TokenType::Identifier, "method name");
184 while (match(TokenType::Dot)) {
185 className += "." + member.text;
186 member = consume(TokenType::Identifier, "method name");
187 }
188 const auto& methodName = member;
189 const auto classLeaf = className.substr(className.find_last_of('.') == std::string::npos
190 ? 0
191 : className.find_last_of('.') + 1);
192 const bool constructor = methodName.text == classLeaf;
193 const auto returnTypeLeaf = returnTypeName.substr(
194 returnTypeName.find_last_of('.') == std::string::npos
195 ? 0
196 : returnTypeName.find_last_of('.') + 1);
197 if (constructor && returnTypeLeaf != classLeaf) {
198 error(methodName, "out-of-line constructor declaration must use its class type");
199 }
200 if (methodName.text == "destroy") {
201 error(methodName, "destructors cannot be defined out-of-line");
202 }
203
204 OutOfLineMethodDefinition definition;
205 definition.className = std::move(className);
206 definition.namespacePath = namespacePath_;
207 definition.location = returnType.location;
208 auto& method = definition.method;
209 method.name = methodName.text;
210 method.returnType = constructor ? "void" : returnTypeName;
211 method.location = returnType.location;
212 method.constructor = constructor;
213 method.parameters = parseParameters();
214 skipNewlines();
215 if (match(TokenType::From)) {
216 const auto symbol = consume(TokenType::String, "external symbol string literal");
217 method.externalBinding = true;
218 method.externalSymbol = symbol.text;
219 consumeStatementTerminator();
220 } else if (check(TokenType::LeftBrace)) {
221 method.body = parseBlock();
222 } else {
223 error(current(), "expected method body or 'from \"<symbol>\"'");
224 }
225 return definition;
226}

228std::string Parser::parseType(bool allowVoid) {
229 const NestingGuard nesting(*this);
230 std::string type;
231 if (match(TokenType::CallbackType)) {
232 consume(TokenType::Less, "'<' after callback");
233 type = "callback<" + parseType(true);
234 consume(TokenType::LeftParen, "'(' before callback parameter types");
235 type += "(";
236 if (!check(TokenType::RightParen)) {
237 do {
238 if (type.back() != '(') type += ",";
239 type += parseType();
240 } while (match(TokenType::Comma));
241 }
242 consume(TokenType::RightParen, "')' after callback parameter types");
243 consume(TokenType::Greater, "'>' after callback signature");
244 type += ")>";
245 }
246 else if (match(TokenType::Int)) type = "int";
247 else if (match(TokenType::Bool)) type = "bool";
248 else if (match(TokenType::FloatType)) type = "float";
249 else if (match(TokenType::Unsigned)) type = "unsigned";
250 else if (match(TokenType::StrgType)) type = "String";
251 else if (match(TokenType::ListType)) type = "list";
252 else if (match(TokenType::DictType)) type = "dict";
253 else if (match(TokenType::BufferType)) type = "buffer";
254 else if (match(TokenType::HandleType)) type = "handle";
255 else if (match(TokenType::AnyType)) type = "any";
256 else if (match(TokenType::TypeType)) type = "type";
257 else if (allowVoid && match(TokenType::Void)) type = "void";
258 else if (check(TokenType::Identifier)) {
259 type = tokens_[current_++].text;
260 while (match(TokenType::Dot)) {
261 type += "." + consume(TokenType::Identifier, "type name component").text;
262 }
263 }

283MethodDeclaration Parser::parseMethod(const Token& typeOrName, const Token& methodName,
284 bool constructor, const std::string& returnType) {
285 MethodDeclaration method;
286 method.name = methodName.text;
287 method.returnType = constructor ? "void" : !returnType.empty() ? returnType :
288 (typeOrName.type == TokenType::Int ? "int" :
289 typeOrName.type == TokenType::Bool ? "bool" :
290 typeOrName.type == TokenType::FloatType ? "float" :
291 typeOrName.type == TokenType::Unsigned ? "unsigned" :
292 typeOrName.type == TokenType::StrgType ? "String" :
293 typeOrName.type == TokenType::Void ? "void" : typeOrName.text);
294 method.location = typeOrName.location;
295 method.constructor = constructor;
296 method.parameters = parseParameters();
297 if (check(TokenType::LeftBrace)) {
298 method.body = parseBlock();
299 } else {
300 method.declarationOnly = true;
301 consumeStatementTerminator();
302 }
303 return method;
