56static void (*find_run_method(const CwhipClassMeta *metadata))(void *) {
57 if (metadata == NULL) return NULL;
58 for (uint64_t index = 0; index < metadata->method_count; ++index) {
59 const CwhipMethodMeta *candidate = &metadata->methods[index];
60 if (candidate->name_length == 3 &&
61 memcmp(candidate->name, "run", 3) == 0) {
62 return (void (*)(void *))candidate->code;
63 }
64 }
65 return NULL;
66}

69static void *cwhip_thread_trampoline(void *raw_args) {
70 CwhipThreadStartArgs *args = (CwhipThreadStartArgs *)raw_args;
71 void *receiver = args->receiver;
72 void (*run_thunk)(void *) = args->run_thunk;
73
74 /* Registers this OS thread with the collector and acquires the global
75 * lock (see cwhip_runtime_thread_enter's doc comment): from this point
76 * on this thread is a full participant in the single-lock threading
77 * model and may safely run Cwhip code / touch the managed heap. */
78 cwhip_runtime_thread_enter();
79 CwhipRootFrame frame = {0};
80 void *const slots[1] = {&receiver};
81 cwhip_gc_push_or_abort(&frame, slots, 1);
82
83 /* Hand off root-protection of the receiver to this thread's own frame
84 * before signaling the spawner, so there is never a window where
85 * neither thread's root frame protects it (see cwhip_thread_start). */
86 pthread_mutex_lock(&args->ready_mutex);
87 args->ready = 1;
88 pthread_cond_signal(&args->ready_cond);
89 pthread_mutex_unlock(&args->ready_mutex);
90
91 run_thunk(receiver);
92
93 cwhip_gc_pop_or_abort(&frame);
94 cwhip_runtime_thread_exit();
95 return NULL;
96}

98void *cwhip_thread_start(void *receiver) {
99 if (receiver == NULL) abort();
100 /* `receiver` is whatever subobject view the compiler passed for this
101 * (statically Thread-typed) call; its own metadata field only lists
102 * Thread's own declared methods. cwhip_gc_root resolves to the true,
103 * most-derived allocated object, whose metadata table is the merged
104 * vtable that actually includes the subclass's "run" method. */
105 void *root = cwhip_gc_root(receiver);
106 const CwhipClassMeta *metadata = *(const CwhipClassMeta *const *)root;
107 void (*run_thunk)(void *) = find_run_method(metadata);
108 if (run_thunk == NULL) abort();
109
110 CwhipThreadStartArgs *args = (CwhipThreadStartArgs *)malloc(sizeof(*args));
111 if (args == NULL) abort();
112 args->receiver = root;
113 args->run_thunk = run_thunk;
114 args->ready = 0;
115 if (pthread_mutex_init(&args->ready_mutex, NULL) != 0) abort();
116 if (pthread_cond_init(&args->ready_cond, NULL) != 0) abort();
117
118 /* Root the object from this (spawning) thread until the new thread
119 * has rooted it on its own stack (see cwhip_thread_trampoline). */
120 CwhipRootFrame frame = {0};
121 void *const slots[1] = {&root};
122 cwhip_gc_push_or_abort(&frame, slots, 1);
123
124 CwhipThreadHandle *handle = (CwhipThreadHandle *)malloc(sizeof(*handle));
125 if (handle == NULL) abort();
126
127 if (pthread_create(&handle->thread, NULL, cwhip_thread_trampoline, args) != 0) {
128 cwhip_gc_pop_or_abort(&frame);
129 pthread_mutex_destroy(&args->ready_mutex);
130 pthread_cond_destroy(&args->ready_cond);
131 free(args);
132 free(handle);
133 abort();
134 }
135
136 /* Wait for the child to take over root-protection of the receiver.
137 * Release the global lock first: the child cannot make progress (it
138 * needs the same lock to register itself) while this thread holds it. */
139 cwhip_runtime_gil_release();
140 pthread_mutex_lock(&args->ready_mutex);
141 while (!args->ready) {
142 pthread_cond_wait(&args->ready_cond, &args->ready_mutex);
143 }
144 pthread_mutex_unlock(&args->ready_mutex);
145 cwhip_runtime_gil_acquire();
146
147 cwhip_gc_pop_or_abort(&frame);
148 pthread_mutex_destroy(&args->ready_mutex);
149 pthread_cond_destroy(&args->ready_cond);
150 free(args);
151 return handle;
152}

154void cwhip_thread_join(void *receiver, void *worker) {
155 (void)receiver;
156 if (worker == NULL) abort();
157 CwhipThreadHandle *handle = (CwhipThreadHandle *)worker;
158 /* Release the global lock so the joined thread (and any other thread)
159 * can keep making progress while this thread blocks. */
160 cwhip_runtime_gil_release();
161 const int joined = pthread_join(handle->thread, NULL);
162 cwhip_runtime_gil_acquire();
163 if (joined != 0) abort();
164 free(handle);
165}

179void cwhip_semaphore_wait(void *receiver, void *semaphore) {
180 (void)receiver;
181 if (semaphore == NULL) abort();
182 CwhipSemaphoreObject *sem = (CwhipSemaphoreObject *)semaphore;
183 /* The semaphore's own mutex/condvar synchronize its count independent
184 * of the global lock; releasing the global lock here only prevents a
185 * blocked wait from starving every other Cwhip thread, exactly like
186 * cwhip_thread_join above. */
187 cwhip_runtime_gil_release();
188 pthread_mutex_lock(&sem->mutex);
189 while (sem->count <= 0) {
190 pthread_cond_wait(&sem->cond, &sem->mutex);
191 }
192 --sem->count;
193 pthread_mutex_unlock(&sem->mutex);
194 cwhip_runtime_gil_acquire();
195}

217/*
218 * Mutex and Condition are *logical* locks: the native guard mutexes below
219 * only protect a few fields for a short, non-blocking critical section and
220 * are never held while acquiring the global lock (GIL) or while running
221 * Cwhip code. Blocking waits happen with the GIL released, so a thread that
222 * owns a logical mutex can always reacquire the GIL and unlock it.
223 */
224static int mutex_owned_by_caller(const CwhipMutexObject *mutex) {
225 return mutex->locked && pthread_equal(mutex->owner, pthread_self());
226}

228void *cwhip_mutex_create(void *receiver) {
229 (void)receiver;
230 CwhipMutexObject *mutex = (CwhipMutexObject *)malloc(sizeof(*mutex));
231 if (mutex == NULL) {
232 cwhip_runtime_set_error(ENOMEM);
233 return NULL;
234 }
235 int result = pthread_mutex_init(&mutex->guard, NULL);
236 if (result != 0) {
237 free(mutex);
238 cwhip_runtime_set_error(result);
239 return NULL;
240 }
241 result = pthread_cond_init(&mutex->available, NULL);
242 if (result != 0) {
243 pthread_mutex_destroy(&mutex->guard);
244 free(mutex);
245 cwhip_runtime_set_error(result);
246 return NULL;
247 }
248 mutex->locked = 0;
249 mutex->waiters = 0;
250 cwhip_runtime_clear_error();
251 return mutex;
252}

254/* Caller holds mutex->guard and not the GIL. */
255static void mutex_acquire_locked_guard(CwhipMutexObject *mutex) {
256 ++mutex->waiters;
257 while (mutex->locked) {
258 pthread_cond_wait(&mutex->available, &mutex->guard);
259 }
260 --mutex->waiters;
261 mutex->locked = 1;
262 mutex->owner = pthread_self();
263}

265int32_t cwhip_mutex_lock(void *receiver, void *object) {
266 (void)receiver;
267 if (object == NULL) {
268 cwhip_runtime_set_error(EINVAL);
269 return 0;
270 }
271 CwhipMutexObject *mutex = (CwhipMutexObject *)object;
272 pthread_mutex_lock(&mutex->guard);
273 if (mutex_owned_by_caller(mutex)) {
274 pthread_mutex_unlock(&mutex->guard);
275 cwhip_runtime_set_error(EDEADLK);
276 return 0;
277 }
278 if (!mutex->locked) {
279 /* Uncontended fast path: no need to give up the GIL. */
280 mutex->locked = 1;
281 mutex->owner = pthread_self();
282 pthread_mutex_unlock(&mutex->guard);
283 cwhip_runtime_clear_error();
284 return 1;
285 }
286 pthread_mutex_unlock(&mutex->guard);
287
288 cwhip_runtime_gil_release();
289 pthread_mutex_lock(&mutex->guard);
290 mutex_acquire_locked_guard(mutex);
291 pthread_mutex_unlock(&mutex->guard);
292 cwhip_runtime_gil_acquire();
293 cwhip_runtime_clear_error();
294 return 1;
295}

370int32_t cwhip_condition_wait(void *receiver, void *condition_object,
371 void *mutex_object) {
372 (void)receiver;
373 if (condition_object == NULL || mutex_object == NULL) {
374 cwhip_runtime_set_error(EINVAL);
375 return 0;
376 }
377 CwhipConditionObject *condition = (CwhipConditionObject *)condition_object;
378 CwhipMutexObject *mutex = (CwhipMutexObject *)mutex_object;
379
380 pthread_mutex_lock(&mutex->guard);
381 const int owned = mutex_owned_by_caller(mutex);
382 pthread_mutex_unlock(&mutex->guard);
383 if (!owned) {
384 cwhip_runtime_set_error(EPERM);
385 return 0;
386 }
387
388 /* Register as a waiter and snapshot the signal sequence while the
389 * logical mutex is still held: any signal issued after the caller
390 * releases the mutex bumps the sequence, so it cannot be lost. */
391 pthread_mutex_lock(&condition->guard);
392 const uint64_t observed = condition->sequence;
393 ++condition->waiters;
394 pthread_mutex_unlock(&condition->guard);
395
396 cwhip_runtime_gil_release();
397
398 pthread_mutex_lock(&mutex->guard);
399 mutex_release_locked_guard(mutex);
400 pthread_mutex_unlock(&mutex->guard);
401
402 pthread_mutex_lock(&condition->guard);
403 while (condition->sequence == observed) {
404 pthread_cond_wait(&condition->condition, &condition->guard);
405 }
406 --condition->waiters;
407 pthread_mutex_unlock(&condition->guard);
408
409 pthread_mutex_lock(&mutex->guard);
410 mutex_acquire_locked_guard(mutex);
411 pthread_mutex_unlock(&mutex->guard);
412
413 cwhip_runtime_gil_acquire();
414 cwhip_runtime_clear_error();
415 return 1;
416}

418static int32_t condition_notify(void *object, int broadcast) {
419 if (object == NULL) {
420 cwhip_runtime_set_error(EINVAL);
421 return 0;
422 }
423 CwhipConditionObject *condition = (CwhipConditionObject *)object;
424 pthread_mutex_lock(&condition->guard);
425 if (condition->waiters > 0) {
426 ++condition->sequence;
427 if (broadcast) pthread_cond_broadcast(&condition->condition);
428 else pthread_cond_signal(&condition->condition);
429 }
430 pthread_mutex_unlock(&condition->guard);
431 cwhip_runtime_clear_error();
432 return 1;
433}
